import React, { useState } from 'react';
import UploadFileTab from './dataset/UploadFileTab';
import PasteDataTab from './dataset/PasteDataTab';
import ConnectApiDbTab from './dataset/ConnectApiDbTab';

function AddDatasetModal({ onClose }) {
  const [activeTab, setActiveTab] = useState('upload');

  return (
    <div className="fixed inset-0 bg-black/50 backdrop-blur-sm flex items-center justify-center z-50">
      <div className="bg-cyber-dark border border-cyber-cyan/30 rounded-lg w-full max-w-2xl shadow-lg">

          <h2 className="text-xl font-bold text-cyber-text">Add New Dataset</h2>
          <button onClick={onClose} className="text-cyber-text hover:text-cyber-cyan">
            &times;
          </button>
        </div>
        
        <div className="p-4">
          <div className="flex border-b border-cyber-detail mb-4">
            <button 
              onClick={() => setActiveTab('upload')} 
              className={`px-4 py-2 ${activeTab === 'upload' ? 'border-b-2 border-cyber-cyan text-cyber-cyan' : 'text-cyber-text'}`}
            >
              Upload File
            </button>
            <button 
              onClick={() => setActiveTab('paste')} 
              className={`px-4 py-2 ${activeTab === 'paste' ? 'border-b-2 border-cyber-cyan text-cyber-cyan' : 'text-cyber-text'}`}
            >
              Paste Data
            </button>
            <button 
              onClick={() => setActiveTab('connect')} 
              className={`px-4 py-2 ${activeTab === 'connect' ? 'border-b-2 border-cyber-cyan text-cyber-cyan' : 'text-cyber-text'}`}
            >
              Connect API/DB
            </button>
          </div>
          
          {activeTab === 'upload' && <UploadFileTab />}
          {activeTab === 'paste' && <PasteDataTab />}
          {activeTab === 'connect' && <ConnectApiDbTab />}

            <button onClick={onClose} className="px-4 py-2 text-cyber-text bg-cyber-detail/50 rounded hover:bg-cyber-detail transition-colors">
              Cancel
            </button>
            <button className="cyber-button">
              Create Dataset
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}

// Example function to fetch datasets
async function fetchDatasets() {
  try {
    const response = await fetch('http://localhost:8000/api/datasets/');
    if (!response.ok) {
      throw new Error('Network response was not ok');
    }
    return await response.json();
  } catch (error) {
    console.error('Error fetching datasets:', error);
    throw error;
  }
}

DatasetListWithMetrics
Este componente muestra una lista de datasets con gráficos de métricas en miniatura:

tsx

Copy
function DatasetListWithMetrics() {
  const datasetList = Array.from({ length: 5 }, (_, i) => ({
    name: `Dataset ${i + 1}`,
    metrics: Array.from({ length: 5 }, () => ({
      value: Math.floor(Math.random() * 50) + 10,
    })),
  }));

  return (
    <div className="bg-black/70 p-4 rounded shadow border border-gray-600">
      <h4 className="text-lg font-semibold mb-2 text-white">Recently Used Datasets</h4>
      <div className="space-y-3">
        {datasetList.map((ds, idx) => (
          <div key={idx} className="flex items-center justify-between border-b border-gray-600 pb-2 last:border-none cursor-pointer hover:bg-gray-700/30">
            <div>
              <p className="font-medium text-white">{ds.name}</p>
              <p className="text-sm text-gray-300">Used in {Math.floor(Math.random() * 10) + 1} experiments</p>
            </div>
            <div className="w-40 h-16">
              <ResponsiveContainer width="100%" height="100%">
                <LineChart data={ds.metrics}>
                  <CartesianGrid strokeDasharray="3 3" stroke="#1C3D5A" />
                  <Tooltip formatter={(value: number) => [`${value}`, 'Metric']} cursor={{ stroke: '#1C3D5A' }} />
                  <XAxis hide />
                  <YAxis hide />
                  <Line type="monotone" dataKey="value" stroke="#00E6E6" strokeWidth={2} dot={false} />
                </LineChart>
              </ResponsiveContainer>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
